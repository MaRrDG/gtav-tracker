# Phase 2 — Map and Collectibles Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A signed-in user sees all 255 objectives on the GTA V map, shows and hides categories, marks objectives done from the map or the panel, and sees per-category tallies that persist to their account.

**Architecture:** The map screen is a Server Component that reads the catalog from disk and the user's progress from Supabase, then hands both to a client island. The island owns interaction: Leaflet markers, the panel, optimistic writes to `/api/progress/[id]` with rollback on failure. Leaflet needs `window`, so the map is loaded with `next/dynamic` and `ssr: false`.

**Tech Stack:** Next.js 16, React 19, TypeScript, Leaflet 1.9, `@phosphor-icons/react`, `next/font` for Archivo and Geist, Vitest 4.

**Spec:** `docs/superpowers/specs/2026-08-20-gta5-tracker-design.md`

## Global Constraints

Everything in the phase 1 plan's Global Constraints still applies. In addition:

- **Shape lock:** `border-radius: 0` everywhere. No exceptions, including buttons, inputs, popups and the mobile sheet.
- **Color lock:** one accent, `--primary`, in both themes. No second accent, no protagonist colors, no gradients, no glows.
- **Type roles:** Archivo for headings and category names (uppercase, tight tracking); Geist for interface text; Geist Mono with `font-variant-numeric: tabular-nums` for every number.
- **Icons:** `@phosphor-icons/react` only. Never hand-write an SVG path.
- **Motion budget, exact values:** marker popup 160ms `cubic-bezier(0.23, 1, 0.32, 1)`; tally fill 120ms color only; mobile sheet 300ms `cubic-bezier(0.32, 0.72, 0, 1)`; fly-to 600ms. Nothing else animates. No looping animation anywhere.
- Animate `transform` and `opacity` only. Never `width`, `height`, `top`, `left`, or `transition: all`.
- Every hover transform sits inside `@media (hover: hover) and (pointer: fine)`.
- `prefers-reduced-motion: reduce` keeps opacity and color transitions and drops movement; fly-to becomes an instant jump.
- **Zero em-dash characters** in any user-visible string.
- UI copy stays active and consistent: the button reads `Mark done`, the state reads `Done`, a failed write reads `Could not save. Retry.`

## File Structure

| Path | Responsibility |
| --- | --- |
| `app/page.tsx` | Server Component: session guard, loads catalog and progress, renders the client island |
| `app/tracker.tsx` | Client island: owns progress state, composes map and panel |
| `components/map-view.tsx` | Leaflet map, tile layer, markers, popups, fly-to |
| `components/panel.tsx` | Overall readout, global controls, category list |

one dense surface with shared tokens, and splitting it would scatter the shape and color

import type { Catalog, IndexedCatalog, Objective } from '@/lib/types';

export function indexCatalog(catalog: Catalog): IndexedCatalog {
  const byId = new Map<string, Objective>();
  const byCategory = new Map<string, Objective[]>(
    catalog.categories.map((category) => [category.id, []]),
  );

  for (const objective of catalog.locations) {
    byId.set(objective.id, objective);
    byCategory.get(objective.cat)?.push(objective);
  }

  return { ...catalog, byId, byCategory };
}

import type { ProgressMap } from '@/lib/types';

export function isDone(state: ProgressMap, id: string): boolean {
  return Object.prototype.hasOwnProperty.call(state, id);
}

/** Returns a new map. The caller keeps the previous one, which is what makes rollback cheap. */
export function setDone(
  state: ProgressMap,
  id: string,
  done: boolean,
  now: string,
): ProgressMap {
  if (done) return { ...state, [id]: now };
  const next = { ...state };
  delete next[id];
  return next;
}

export function countDone(state: ProgressMap, ids: string[]): number {
  let total = 0;
  for (const id of ids) if (isDone(state, id)) total += 1;
  return total;
}
```

/** Marks one objective. Rejects with a message the interface can show as it is. */
export async function putObjective(id: string, done: boolean): Promise<void> {
  const response = await fetch(`/api/progress/${encodeURIComponent(id)}`, {
    method: 'PUT',
    headers: { 'content-type': 'application/json' },
    body: JSON.stringify({ done }),
  });

  if (response.status === 401) throw new Error('Session expired. Sign in again.');
  if (!response.ok) throw new Error('Could not save. Retry.');
}
```

Create `components/map-view.tsx`:

```tsx
'use client';

import { useEffect, useRef } from 'react';
import L from 'leaflet';
import 'leaflet/dist/leaflet.css';
import type { Objective, ProgressMap } from '@/lib/types';
import { isDone } from '@/lib/progress-state';

const MIN_ZOOM = 0;
const MAX_ZOOM = 5;
const HOME: { center: L.LatLngExpression; zoom: number } = { center: [72, -122], zoom: 3 };

// A square pin, matching the shape lock. Filled with the accent, hollow once done.
function pinIcon(done: boolean) {
  return L.divIcon({
    className: `pin${done ? ' pin--done' : ''}`,
    html: '<span class="pin__body"></span>',
    iconSize: [14, 14],
    iconAnchor: [7, 7],
    popupAnchor: [0, -8],
  });
}

type Props = {
  objectives: Objective[];
  visible: Set<string>;
  progress: ProgressMap;
  hideCompleted: boolean;
  focusId: string | null;
  onToggle: (id: string) => void;
};

export default function MapView({
  objectives,
  visible,
  progress,
  hideCompleted,
  focusId,
  onToggle,
}: Props) {
  const container = useRef<HTMLDivElement>(null);
  const map = useRef<L.Map | null>(null);
  const markers = useRef(new Map<string, L.Marker>());
  const layers = useRef(new Map<string, L.LayerGroup>());
  // Kept in a ref so rebuilding markers is not needed when the handler identity changes.
  const toggle = useRef(onToggle);
  toggle.current = onToggle;

  // Create the map once.
  useEffect(() => {
    if (!container.current || map.current) return;

    map.current = L.map(container.current, {
      minZoom: MIN_ZOOM,
      maxZoom: MAX_ZOOM,
      attributionControl: false,
    }).setView(HOME.center, HOME.zoom);

    L.tileLayer('/tiles/{z}/{x}/{y}.jpg', {
      minZoom: MIN_ZOOM,
      maxZoom: MAX_ZOOM,
      noWrap: true,
    }).addTo(map.current);

    return () => {
      map.current?.remove();
      map.current = null;
      markers.current.clear();
      layers.current.clear();
    };
  }, []);

  // Build one marker per objective, once.
  useEffect(() => {
    if (!map.current) return;

    for (const objective of objectives) {
      if (objective.lat === undefined || objective.lng === undefined) continue;
      if (markers.current.has(objective.id)) continue;

      const marker = L.marker([objective.lat, objective.lng], { icon: pinIcon(false) });
      marker.bindPopup(() => popupHtml(objective), { closeButton: false });
      marker.on('popupopen', (event) => {
        const button = event.popup.getElement()?.querySelector('[data-toggle]');
        button?.addEventListener('click', () => {
          toggle.current(objective.id);
          marker.closePopup();
        });
      });

      markers.current.set(objective.id, marker);

      let layer = layers.current.get(objective.cat);
      if (!layer) {
        layer = L.layerGroup();
        layers.current.set(objective.cat, layer);
      }
      layer.addLayer(marker);
    }
  }, [objectives]);

  // Reflect visibility, completion and the hide-completed switch.
  useEffect(() => {
    const instance = map.current;
    if (!instance) return;

    for (const [cat, layer] of layers.current) {
      if (visible.has(cat)) layer.addTo(instance);
      else instance.removeLayer(layer);
    }

    for (const objective of objectives) {
      const marker = markers.current.get(objective.id);
      if (!marker) continue;

      const done = isDone(progress, objective.id);
      marker.setIcon(pinIcon(done));

      const layer = layers.current.get(objective.cat);
      if (!layer) continue;
      const shouldShow = !(done && hideCompleted);
      if (shouldShow && !layer.hasLayer(marker)) layer.addLayer(marker);
      if (!shouldShow && layer.hasLayer(marker)) layer.removeLayer(marker);
    }
  }, [objectives, visible, progress, hideCompleted]);

  // Fly to the objective the panel asked for.
  useEffect(() => {
    const instance = map.current;
    if (!instance || !focusId) return;

    const objective = objectives.find((item) => item.id === focusId);
    const marker = markers.current.get(focusId);
    if (!objective || objective.lat === undefined || objective.lng === undefined) return;

    const target: L.LatLngExpression = [objective.lat, objective.lng];
    const zoom = Math.max(instance.getZoom(), 4);
    const still = window.matchMedia('(prefers-reduced-motion: reduce)').matches;

    if (still) instance.setView(target, zoom);
    else instance.flyTo(target, zoom, { duration: 0.6 });

    marker?.openPopup();
  }, [focusId, objectives]);

  return <div ref={container} className="map" />;
}

function popupHtml(objective: Objective): string {
  const notes = objective.notes ? `<p class="popup__notes">${escapeHtml(objective.notes)}</p>` : '';
  return `<div class="popup">
    <h2 class="popup__name">${escapeHtml(objective.name)}</h2>
    ${notes}
    <button type="button" data-toggle>Mark done</button>
  </div>`;
}

// The catalog is ours, but it is still data flowing into innerHTML.
function escapeHtml(value: string): string {
  return value
    .replaceAll('&', '&amp;')
    .replaceAll('<', '&lt;')
    .replaceAll('>', '&gt;')
    .replaceAll('"', '&quot;');
}
```

The popup button always reads `Mark done` because the popup only opens from a marker the
user is about to act on; marking an already-done objective is done from its tally cell.
If that proves wrong in use, the popup can read the completion state and switch the label.

Create `app/tracker.tsx`:

```tsx
'use client';

import dynamic from 'next/dynamic';
import { useCallback, useMemo, useState } from 'react';
import type { Catalog, ProgressMap } from '@/lib/types';
import { indexCatalog } from '@/lib/catalog-index';
import { setDone, isDone } from '@/lib/progress-state';
import { putObjective } from '@/lib/progress-client';

// Leaflet reaches for window at import time, so it never renders on the server.
const MapView = dynamic(() => import('@/components/map-view'), { ssr: false });

type Props = {
  catalog: Catalog;
  initialProgress: ProgressMap;
  email: string;
};

export function Tracker({ catalog, initialProgress, email }: Props) {
  const indexed = useMemo(() => indexCatalog(catalog), [catalog]);
  const [progress, setProgress] = useState(initialProgress);
  const [visible, setVisible] = useState(
    () => new Set(catalog.categories.map((category) => category.id)),
  );
  const [hideCompleted, setHideCompleted] = useState(false);
  const [focusId, setFocusId] = useState<string | null>(null);
  const [error, setError] = useState<string | null>(null);

  const toggle = useCallback(async (id: string) => {
    const now = new Date().toISOString();
    let previous: ProgressMap = {};
    let next = false;

    setProgress((current) => {
      previous = current;
      next = !isDone(current, id);
      return setDone(current, id, next, now);
    });

    setError(null);

    try {
      await putObjective(id, next);
    } catch (failure) {
      // The interface never claims a save that did not happen.
      setProgress(previous);
      setError(failure instanceof Error ? failure.message : 'Could not save. Retry.');
    }
  }, []);

  return (
    <div className="layout">
      {/* The panel arrives in Task 4. */}
      <aside className="panel">
        <p>{email}</p>
        {error && <p role="alert">{error}</p>}
      </aside>
      <MapView
        objectives={catalog.locations}
        visible={visible}
        progress={progress}
        hideCompleted={hideCompleted}
        focusId={focusId}
        onToggle={toggle}
      />
    </div>
  );
}
```

- [ ] **Step 4: Load catalog and progress on the server**

Replace `app/page.tsx`:

```tsx
import { redirect } from 'next/navigation';
import { createServerSupabase } from '@/lib/supabase/server';
import { getCatalog } from '@/lib/catalog';
import type { ProgressMap } from '@/lib/types';
import { Tracker } from './tracker';

export default async function HomePage() {
  const supabase = await createServerSupabase();
  const {
    data: { user },
  } = await supabase.auth.getUser();
  if (!user) redirect('/sign-in');

  // Read straight from Supabase rather than through /api/progress: this is already the
  // server, and a request to our own API would only add a round trip. /api stays the
  // entry point for the browser and for any other client.
  const { data } = await supabase.from('progress').select('objective_id, completed_at');

  const initialProgress: ProgressMap = {};
  for (const row of data ?? []) initialProgress[row.objective_id] = row.completed_at;

  return (
    <Tracker
      catalog={getCatalog()}
      initialProgress={initialProgress}
      email={user.email ?? ''}
    />
  );
}

'use client';

import type { Objective, ProgressMap } from '@/lib/types';
import { isDone } from '@/lib/progress-state';

type Props = {
  objectives: Objective[];
  progress: ProgressMap;
  onFocus: (objectiveId: string) => void;
};

/**
 * One cell per objective. Shows not just how many are missing but which, and clicking a
 * cell takes the map to it. This replaces a progress bar rather than sitting beside one.
 */
export function Tally({ objectives, progress, onFocus }: Props) {
  return (
    <div className="tally">
      {objectives.map((objective) => {
        const done = isDone(progress, objective.id);
        return (
          <button
            key={objective.id}
            type="button"
            className={`tally__cell${done ? ' tally__cell--done' : ''}`}
            aria-label={`${objective.name}. ${done ? 'Done' : 'Not done'}. Show on map.`}
            onClick={() => onFocus(objective.id)}
          />
        );
      })}
    </div>
  );
}
```
